# New game -> the character creator (Enter through it) -> walk into the start village inn -> talk to the innkeeper -> open the shop, buy -> walk out.
#   embervale.exe --script tools/scripts/inn.txt        (screenshots land in the working directory)
# Times are script seconds; script time stops while a walkto runs. Exit code 3 if an expect/walkto fails.
seed 7
0.8  shot inn_0_title.png
1.0  key Return                  # title: New Game (test runs never see the Continue entry) opens the creator
1.2  expect mode creator
1.3  shot inn_0b_creator.png
1.4  key Return                  # LOOKS -> NAME
1.5  key Return                  # NAME -> ORIGIN
1.6  key Return                  # ORIGIN -> HOMELAND (M3)
1.7  key Return                  # HOMELAND: BEGIN
1.8  expect mode play
2.5  shot inn_1_start.png
2.6  walkto inn                  # steers with held W/A/S/D along a tile path, then walks in the door
3.4  expect inside 1
3.5  shot inn_2_inside.png
3.6  walkto innkeeper
3.8  key E                       # talk
4.0  expect mode dialogue
6.0  shot inn_3_greeting.png     # the greeting has finished typing out by now
6.2  key Down                    # option 0 is the job offer on a fresh game, 1 is the wares
6.4  key Return
6.6  expect mode shop
7.2  shot inn_4_shop.png
7.4  key Return                  # buy the first item
8.0  shot inn_5_bought.png
8.2  key Escape                  # close the shop
8.4  expect mode play
8.5  walkto exit
9.5  expect inside 0
10.0 shot inn_6_outside.png
10.2 quit
